-- Normalize stock_transactions approval states for purchase flow:
-- 0 = IN_PROGRESS, 2 = WAITING_APPROVAL, 1 = APPROVED
UPDATE stock_transactions st
JOIN purchase_loads pl ON pl.id = st.reference_id
SET st.isApproved = 2
WHERE st.type = 'PURCHASE'
  AND st.isApproved = 0
  AND pl.status = 'PENDING';
